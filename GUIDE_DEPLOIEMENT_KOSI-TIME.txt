PARTIE 7 : CONFIGURATION APACHE2 (PORT 8081)
================================================================================

# Éditer le fichier ports.conf

# Sauvegarder et quitter


# Créer le fichier de configuration du VirtualHost
sudo nano /etc/apache2/sites-available/kosi-time.conf

# ----------- CONTENU DU FICHIER -----------
<VirtualHost *:8081>
    ServerName kosi-time.run-telemat.com
    DocumentRoot /var/www/admin-kosi-pointage/public

    <Directory /var/www/admin-kosi-pointage>
        AllowOverride All
    </Directory>

    <Directory /var/www/admin-kosi-pointage/public>
        Options -Indexes +FollowSymLinks
        AllowOverride All
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/kosi-time-error.log
    CustomLog ${APACHE_LOG_DIR}/kosi-time-access.log combined
</VirtualHost>
# ------------------------------------------

# Sauvegarder et quitter


# Activer le site et le module rewrite
sudo a2ensite kosi-time.conf
sudo a2enmod rewrite
sudo systemctl restart apache2

PARTIE 8 : CONFIGURATION NGINX (REVERSE PROXY + SSL)
================================================================================

# Créer le fichier de configuration Nginx
sudo nano /etc/nginx/sites-available/kosi-time

# ----------- CONTENU DU FICHIER -----------
# Redirection HTTP vers HTTPS
server {
    listen 80;
    server_name kosi-time.run-telemat.com;
    return 301 https://$host$request_uri;
}

# Configuration HTTPS
server {
    listen 443 ssl http2;
    server_name kosi-time.run-telemat.com;

    # Certificats SSL (seront générés par Certbot)
    ssl_certificate /etc/letsencrypt/live/kosi-time.run-telemat.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/kosi-time.run-telemat.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Logs
    access_log /var/log/nginx/kosi-time-access.log;
    error_log /var/log/nginx/kosi-time-error.log;

    # Proxy vers Apache2 port 8081
    location / {
        proxy_pass http://127.0.0.1:8081;
        proxy_http_version 1.1;
        
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
}
# ------------------------------------------

# Sauvegarder et quitter


# Créer un lien symbolique pour activer le site
sudo ln -s /etc/nginx/sites-available/kosi-time /etc/nginx/sites-enabled/

# Tester la configuration Nginx
sudo nginx -t

# Redémarrer Nginx
sudo systemctl restart nginx
